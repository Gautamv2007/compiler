check = (x:int) -> {
  i:int = 2;
  flag:int = 0;
  while(i < x){
    rem:int = x%i;

    if (rem == 0){
      flag = 1;
    }
    i = i + 1;
  }
  return flag;
}

main = () -> {
  x:int = 9;

  out:int = check(x);

  if (out == 0){
    print("The number is prime\n");
  }
  else {
    print("The number is not prime\n");
  }

  return 0;
}
