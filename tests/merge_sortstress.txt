  GVR Stress Test: 1,000 Element Merge Sort
###

merge:void = (arr:int[], l:int, m:int, r:int) -> {
    n1:int = m - l + 1;
    n2:int = r - m;

    # Increased temp arrays to hold half of 1,000
    L:int[500]; 
    R:int[500];

    for(i:int = 0; i < n1; i += 1) {
        L[i] = arr[l + i];
    }
    for(j:int = 0; j < n2; j += 1) {
        R[j] = arr[m + 1 + j];
    }

    i:int = 0; 
    j:int = 0; 
    k:int = l;

    while (i < n1 && j < n2) {
        if (L[i] <= R[j]) {
            arr[k] = L[i];
            i += 1;
        } else {
            arr[k] = R[j];
            j += 1;
        }
        k += 1;
    }

    while (i < n1) {
        arr[k] = L[i];
        i += 1;
        k += 1;
    }

    while (j < n2) {
        arr[k] = R[j];
        j += 1;
        k += 1;
    }

    return 0;
}

mergeSort:void = (arr:int[], l:int, r:int) -> {
    if (l < r) {
        m:int = l + (r - l) / 2;

        mergeSort(arr, l, m);
        mergeSort(arr, m + 1, r);

        merge(arr, l, m, r);
    }
    return 0;
}

main = () -> {
    size:int = 1000;
    arr:int[1000];
    
    # Automatically generate [1000, 999, 998 ... 3, 2, 1]
    for(i:int = 0; i < size; i += 1) {
        arr[i] = size - i;
    }

    for(i:int = 0;i<size;i+=1){
      print(i, " ");
    }
    print("\n");

    return 0;
}
